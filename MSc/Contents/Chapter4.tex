\chapter{مطالعهٔ موردی با داده‌های ساختگی}
\label{chap4}

\section{طراحی مطالعه}

داده‌های ساختگی از جواب دقیق مدل معیار \eqref{eq:test-model} با پارامتر مرجع $\theta^{\ast}=0.8$ تولید می‌شوند:
\begin{equation}
  y_i=\exp(-\theta^{\ast}t_i)+\varepsilon_i,
  \qquad t_i=0,0.1,\ldots,1.
  \label{eq:synthetic-data}
\end{equation}
خطاهای $\varepsilon_i$ مستقل، با میانگین صفر و انحراف معیار $\sigma$ در نظر گرفته شده‌اند. برآورد عددی پارامتر از کمینه‌سازی
\begin{equation}
  \widehat{\theta}_{\Delta t}
  =\argmin_{\theta\in[0.1,2]}
  \frac{1}{m}\sum_{i=1}^{m}
  \left(y_i-x_{\Delta t}(t_i;\theta)\right)^2
  \label{eq:parameter-estimate}
\end{equation}
به‌دست می‌آید؛ $x_{\Delta t}$ پاسخ گسستهٔ مدل است.

\section{تحلیل حساسیت عددی}

برای یک مجموعهٔ دادهٔ ثابت، وابستگی برآورد به گام زمانی با شاخص
\begin{equation}
  S_{\Delta t}=
  \frac{\left\lvert
  \widehat{\theta}_{\Delta t/2}-\widehat{\theta}_{\Delta t}
  \right\rvert}
  {1+\left\lvert\widehat{\theta}_{\Delta t/2}\right\rvert}
  \label{eq:step-sensitivity}
\end{equation}
سنجیده می‌شود. کوچک‌بودن این شاخص، پایداری برآورد نسبت به پالایش گام را نشان می‌دهد؛ بااین‌حال، دربارهٔ کفایت مدل یا سازگاری آن با دادهٔ واقعی حکمی صادر نمی‌کند. اعتبار عددی، کیفیت برازش و اعتبار علمی مدل سه ارزیابی متمایز باقی می‌مانند.
